# Optional extra: semantic/vector search for the MCP server.
# The base install (requirements.txt) stays stdlib-only + the toolkit's mcp[cli] extra;
# the query engine (src/semantic_search.py) lazily imports these and falls back to
# keyword-only search when they're absent. Install only to BUILD the vector index
# (src/build_embeddings.py) or to SERVE semantic queries.
numpy
# Only needed to compute the 2-D topic-map projection (src/build_topic_projection.py, run
# once after a re-embed). The committed projection cache + the HTML generator
# (build_topic_map.py) need neither umap nor numpy.
umap-learn
# CPU-appropriate fallback backend: static (model2vec) embeddings — token-vector lookup,
# no transformer forward pass, so it embeds ~10k texts/sec on CPU. Auto-selected by
# `--backend auto` only when no CUDA GPU is available (see torch below).
model2vec
# Production backend given a GPU (auto-selected by `--backend auto` when
# torch.cuda.is_available()): BAAI/bge-m3, chunk-level embedding. Its 8192-token context
# means a full CHUNK_CHARS chunk embeds whole — the previous default,
# BAAI/bge-large-en-v1.5, capped at 512 tokens and silently truncated ~40% of every
# chunk. bge-m3 is XLM-RoBERTa-large (~2.3 GB); no extra dependency beyond
# sentence-transformers, and only its dense head is used. On a CUDA-capable host, plain
# `pip install torch` resolves a CUDA-enabled wheel (matches your installed driver
# automatically on recent pip/torch). CPU-only hosts should instead use:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
torch
sentence-transformers
